name: Build Windows EXE

on:
  push:
    branches: [ "main" ]
  workflow_dispatch:

jobs:
  test:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: Abhängigkeiten
        run: pip install -r requirements.txt
      - name: Headless-Tests
        env:
          SDL_VIDEODRIVER: dummy
          SDL_AUDIODRIVER: dummy
        run: |
          python tests/test_core.py
          python tests/test_net.py
          python tests/test_net_game.py
          python tests/test_smoke_app.py
          python tests/test_party.py
          python tests/test_party_net.py

  build:
    needs: test
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: Abhängigkeiten installieren
        run: pip install -r requirements.txt
      - name: Assets erzeugen
        run: python tools/make_assets.py
      - name: Bauen (Ordner-Modus)
        run: >
          pyinstaller --noconfirm --onedir --windowed --noupx
          --name "Ru-Curve" --icon "icon.ico" --version-file "version.txt"
          --add-data "assets;assets" --collect-submodules rucurve
          main.py
      - name: Pruefen + ZIP packen
        shell: pwsh
        run: |
          if (-not (Test-Path 'dist/Ru-Curve/Ru-Curve.exe')) { throw "Build kaputt - EXE fehlt" }
          Copy-Item LIESMICH-Download.txt dist/Ru-Curve/LIESMICH.txt
          Compress-Archive -Path dist/Ru-Curve -DestinationPath dist/Ru-Curve.zip -Force
      - name: ZIP als Artifact hochladen
        uses: actions/upload-artifact@v4
        with:
          name: Ru-Curve
          path: dist/Ru-Curve.zip
